mlp_run.c — mlp_init()  ·  Load Weights into Static Instance
mlp_run.c

mlp_init() is called once at startup from main().

It copies the weights from the static const arrays in mlp_weights_data.h (ROM/flash) into the static MLP_Instance g_mlp struct in RAM (BSS segment).

No malloc. No file open. No heap. The struct lives in BSS — zeroed by Zynq boot code. Instance is ready for repeated inference calls.

/* mlp_run.c */

/* Single static instance — no malloc */
static MLP_Instance g_mlp;   // BSS

int mlp_init(void) {
  /* copy ROM arrays → RAM struct */
  for(i=0; i<400; i++)
    for(j=0; j<40; j++)
      g_mlp.fc1_W[i][j] =
          fc1_W[i*40+j];  // from .h

  memcpy(g_mlp.fc1_b, fc1_b, ...);
  /* repeat for fc2, fc3, LN params */
  g_mlp.initialized = 1;
  return 0;
}

main.c — 20 Complex Samples → x[40] Input Vector
main.c

Your 20 complex radar samples are stored as double complex_samples[20][2] in main.c.

build_input() splits each complex number and fills x[40]:

→ x[0..19] = real parts (I channel)

→ x[20..39] = imaginary parts (Q channel)

For live streaming from ADC, replace complex_samples[][] with reads from your DMA / AXI buffer.

/* main.c */

/* 20 radar samples */
double complex_samples[20][2] = {
  { 0.007664, -0.000623 }, // 0
  { 0.007752,  0.000399 }, // 1
  /* ...18 more rows... */
};

void build_input(float x[40]) {
  for(int i=0; i<20; i++){
    x[i]      = (float)samples[i][0]; // real
    x[i+20]  = (float)samples[i][1]; // imag
  }
}

mlp_run.c — mlp_forward()  ·  3-Layer Forward Pass
mlp_run.c

mlp_forward(x, y) runs the full network. All intermediate buffers (h1, h2) are on the stack (~9.6 KB). No heap used.

Layer 1: Linear(40→400) + LayerNorm + ReLU

Layer 2: Linear(400→400) + LayerNorm + ReLU

Layer 3: Linear(400→2) — no activation

Formula: y = fc3(LN₂(fc2(LN₁(fc1(x)))))

/* mlp_run.c */

void mlp_forward(
    const float x[40], float y[2])
{
  float h1[400], h2[400]; // stack

  /* fc1 → LayerNorm → ReLU */
  linear_layer(x,  40,  h1_pre, 400, ...);
  layer_norm  (h1_pre, h1_ln, 400, ...);
  for(i) h1[i] = relu_f(h1_ln[i]);

  /* fc2 → LayerNorm → ReLU */
  linear_layer(h1, 400, h2_pre, 400, ...);
  layer_norm  (h2_pre, h2_ln, 400, ...);
  for(i) h2[i] = relu_f(h2_ln[i]);

  /* fc3 → y[2] */
  linear_layer(h2, 400, y, 2, ...);
}

main.c — Print y[0], y[1] via UART (xil_printf)
main.c

# Zynq-7000 (Cortex-A9)
arm-none-eabi-gcc \
  -mcpu=cortex-a9 -mfpu=vfpv3 -mfloat-abi=hard \
  -O2 -Wall -std=c99 \
  -I. \
  main.c mlp_run.c \
  -lm \
  -o doppler_mlp.elf

# Zynq UltraScale+ (Cortex-A53)

  -mcpu=cortex-a53 -O2 -std=c99 \
  -I. \
  main.c mlp_run.c \
  -lm \
  -o doppler_mlp.elf
